**1. 历史消息字符数限制：**
```typescript
const MAX_HISTORY_CHARS = 8000;
let totalChars = 0;
const history: { role: ...; content: string }[] = [];
for (let i = messages.length - 1; i >= 0; i--) {
  const m = messages[i];
  if (!m.content.trim() || m.content.startsWith("> ⚠️") || m.content.startsWith("> （")) continue;
  if (totalChars + m.content.length > MAX_HISTORY_CHARS) break;
  history.unshift({ role: m.role, content: m.content });
  totalChars += m.content.length;
  if (history.length >= 10) break;
}
```
