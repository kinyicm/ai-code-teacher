```typescript
export interface ParsedError {
  isJavaStack: boolean;
  exceptionType?: string;
  exceptionSimple?: string;
  message?: string;
  thread?: string;
  locations: { file: string; line?: number; method?: string }[];
  summary: string;
}

const EXCEPTION_GLOSSARY: Record<string, string> = {
  ArrayIndexOutOfBoundsException: "数组下标越界：访问了数组不存在的位置",
  NullPointerException: "空指针异常：对一个值为null的对象调用了方法",
  NumberFormatException: "数字格式错误：把非数字字符串转成了数字",
  ArithmeticException: "算术异常：最常见的是除以 0",
  ClassCastException: "类型转换错误：把对象强行转成了它不是的类型",
  IllegalArgumentException: "参数不合法：传给方法的参数不符合要求",
  IllegalStateException: "状态不合法：在错误的时机调用了方法",
  IndexOutOfBoundsException: "索引越界：访问了集合不存在的位置",
  StringIndexOutOfBoundsException: "字符串索引越界",
  OutOfMemoryError: "内存不足：程序占用内存超过了JVM上限",
  StackOverflowError: "栈溢出：通常是递归没有终止条件",
  FileNotFoundException: "文件不存在：指定路径的文件找不到",
  IOException: "输入输出异常：读写文件或网络时出错",
  InputMismatchException: "输入类型不匹配：Scanner读取的内容类型不对",
  NoSuchElementException: "没有更多元素：Scanner或Iterator已读完还继续取",
  ConcurrentModificationException: "并发修改异常：遍历集合时同时修改了集合",
};

export function parseJavaError(input: string): ParsedError {
  const text = input.trim();
  const result: ParsedError = { isJavaStack: false, locations: [], summary: "" };

  const hasException = /\b(\w+\.)*(\w+Exception|\w+Error)\b/.test(text);
  const hasAtLine = /^\s*at\s+/m.test(text);
  if (!hasException && !hasAtLine) return result;

  result.isJavaStack = true;

  const threadMatch = text.match(/Exception in thread\s+"([^"]+)"/);
  if (threadMatch) result.thread = threadMatch[1];

  const excMatch = text.match(/((?:[\w$]+\.)*([\w$]+(?:Exception|Error)))(?::\s*([^\n]+))?/);
  if (excMatch) {
    result.exceptionType = excMatch[1];
    result.exceptionSimple = excMatch[2];
    if (excMatch[3]) result.message = excMatch[3].trim();
  }

  const atRegex = /^\s*at\s+([\w$.]+)\.([\w$<>]+)\(([^:)]+)(?::(\d+))?\)/gm;
  let m: RegExpExecArray | null;
  while ((m = atRegex.exec(text)) !== null) {
    result.locations.push({
      file: m[3],
      line: m[4] ? parseInt(m[4], 10) : undefined,
      method: `${m[1]}.${m[2]}`,
    });
  }

  const parts: string[] = [];
  if (result.exceptionSimple) {
    const gloss = EXCEPTION_GLOSSARY[result.exceptionSimple];
    parts.push(`异常类型：${result.exceptionSimple}${gloss ? `（${gloss}）` : ""}`);
  }
  if (result.message) parts.push(`错误信息：${result.message}`);
  if (result.locations.length > 0) {
    const first = result.locations[0];
    parts.push(`出错位置：${first.file}${first.line ? ` 第 ${first.line} 行` : ""}`);
  }
  if (result.thread) parts.push(`发生线程：${result.thread}`);
  result.summary = parts.join("\n");

  return result;
}

export function looksLikeErrorStack(input: string): boolean {
  const text = input.trim();
  return /\b\w+Exception\b/.test(text) || /\b\w+Error\b/.test(text) ||
    /^\s*at\s+/m.test(text) || /Exception in thread/.test(text);
}
```
